stages:

variables:
  GIT_DEPTH: 2

include:

scan-update-set:
  stage: scan
  image: alpine:latest
  before_script:
    - apk add --no-cache git
  script:
    - |
      BEFORE_SHA="$CI_COMMIT_BEFORE_SHA"

      if [ -z "$BEFORE_SHA" ] || [ "$BEFORE_SHA" = "0000000000000000000000000000000000000000" ]; then
        echo "No valid before-SHA (first commit on branch) — falling back to HEAD~1"
        BEFORE_SHA=$(git rev-parse HEAD~1 2>/dev/null || echo "")
      fi

      if [ -z "$BEFORE_SHA" ]; then
        echo "No comparison point available - scanning all XML files"
        CHANGED_FILES=$(find update_sets -name '*.xml' 2>/dev/null)
      else
        CHANGED_FILES=$(git diff --name-only "$BEFORE_SHA" "$CI_COMMIT_SHA" -- 'update_sets/' 2>/dev/null | grep '\.xml$' || true)
        if [ -z "$CHANGED_FILES" ]; then
          echo "git diff against $BEFORE_SHA produced nothing usable — scanning all XML files as fallback"
          CHANGED_FILES=$(find update_sets -name '*.xml' 2>/dev/null)
        fi
      fi

      if [ -z "$CHANGED_FILES" ]; then
        echo "No update set XML files changed - nothing to scan"
        exit 0
      fi

      echo "Scanning changed files:"
      printf '%s\n' "$CHANGED_FILES"

      SQUOTE="'"
      SCAN_FAILED=0

      # newline-safe loop — handles filenames with spaces (e.g. "G.U.S.S. v2.1.xml")
      printf '%s\n' "$CHANGED_FILES" | while IFS= read -r file; do
        [ -z "$file" ] && continue
        echo "=== Scanning $file ==="
        if grep -iE "(password|secret|token|api_key)\s*=\s*[\"${SQUOTE}][^\"${SQUOTE}]{4,}" "$file"; then
          echo "FAILED: Possible hardcoded credentials found in $file"
          exit 1
        fi
        echo "PASSED: $file"
      done || SCAN_FAILED=1

      if [ "$SCAN_FAILED" -eq 1 ]; then
        echo "Secret scan failed — aborting"
        exit 1
      fi

      printf '%s\n' "$CHANGED_FILES" > changed_files.txt
  artifacts:
    paths:
      - changed_files.txt
      - update_sets/*.xml
    when: always
  rules:
    - if: '$CI_COMMIT_BRANCH == $CI_DEFAULT_BRANCH'

upload-to-servicenow:
  stage: upload
  image: curlimages/curl:latest
  script:
    - |
      if [ ! -f changed_files.txt ]; then
        echo "No changed_files.txt — nothing to upload"
        exit 0
      fi

      if [ ! -s changed_files.txt ]; then
        echo "changed_files.txt is empty — nothing to upload"
        exit 0
      fi

      # newline-safe loop — handles filenames with spaces
      while IFS= read -r FILE; do
        [ -z "$FILE" ] && continue
        echo "=== Uploading $FILE ==="
        if [ ! -f "$FILE" ]; then
          echo "ERROR: $FILE not found"
          exit 1
        fi
        HTTP_CODE=$(curl -s -o /tmp/sn_response.txt -w "%{http_code}" -X POST \
          -u "$SN_USERNAME:$SN_PASSWORD" \
          -H "Content-Type: application/xml" \
          --data-binary "@$FILE" \
          "$SN_TARGET_INSTANCE/api/global/update_set_processor/import")
        echo "Response: $(cat /tmp/sn_response.txt)"
        echo "HTTP status: $HTTP_CODE"
        if [ "$HTTP_CODE" -lt 200 ] || [ "$HTTP_CODE" -gt 299 ]; then
          echo "ERROR: Upload failed with HTTP $HTTP_CODE"
          exit 1
        fi
        echo "Uploaded $FILE successfully"
      done < changed_files.txt
  rules:
    - if: '$CI_COMMIT_BRANCH == $CI_DEFAULT_BRANCH'
